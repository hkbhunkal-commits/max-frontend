const express = require("express"); const cors = require("cors"); const app = express(); app.use(cors()); app.use(express.json()); const ACCOUNT_ID = process.env.CLOUDFLARE_ACCOUNT_ID; const API_TOKEN = process.env.CLOUDFLARE_API_TOKEN; const MEMORY_API = "https://max-memory-api.hkbhunkal.workers.dev"; // ========================= // HOME // ========================= app.get("/", (req, res) => { res.send("MAX backend is running."); }); // ========================= // TEST // ========================= app.get("/test", (req, res) => { res.json({ status: "OK", message: "MAX backend connection is working!" }); }); // ========================= // GET MEMORIES // ========================= async function getMemories() { try { const response = await fetch( `${MEMORY_API}/memories?user_id=default` ); const data = await response.json(); if (!response.ok || !data.success) { return []; } return data.memories || []; } catch (error) { console.error("Memory read error:", error); return []; } } // ========================= // SAVE MEMORY // ========================= async function saveMemory(memory) { try { const response = await fetch( `${MEMORY_API}/remember`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ user_id: "default", memory: memory }) } ); const data = await response.json(); return response.ok && data.success; } catch (error) { console.error("Memory save error:", error); return false; } } // ========================= // CHAT // ========================= app.post("/chat", async (req, res) => { try { const message = req.body.message; if (!message) { return res.status(400).json({ error: "Message is required." }); } if (!ACCOUNT_ID || !API_TOKEN) { return res.status(500).json({ error: "Cloudflare credentials are missing." }); } // ========================= // REMEMBER COMMAND // ========================= const lowerMessage = message.toLowerCase(); if ( lowerMessage.startsWith("remember ") || lowerMessage.startsWith("remember that ") ) { let memory = message .replace(/^remember that /i, "") .replace(/^remember /i, "") .trim(); if (memory) { const saved = await saveMemory(memory); if (saved) { return res.json({ reply: "Okay bro, I'll remember that." }); } } } // ========================= // LOAD MEMORIES // ========================= const memories = await getMemories(); let memoryText = "No saved memories."; if (memories.length > 0) { memoryText = memories .map((item) => `- ${item.memory}`) .join("\n"); } // ========================= // MAX AI PROMPT // ========================= const prompt = `You are MAX, a natural conversational AI phone assistant. IMPORTANT LANGUAGE RULES: Always reply in the SAME language and style used by the user. If the user speaks Hinglish, reply naturally in Hinglish. Do NOT translate Hinglish into pure Hindi or pure English. If the user speaks Hindi, reply in Hindi. If the user speaks English, reply in English. If the user mixes Hindi and English, keep the same natural Hindi-English mix. Never provide a translation unless the user specifically asks for one. Never say things like: "Here is the Hindi translation." "Here is the English translation." "Translation:" or anything similar. Understand informal spelling, slang and casual speech naturally. For voice conversations, answer naturally like a phone assistant. Keep normal replies SHORT. Usually answer in 1 or 2 sentences. Do not give long explanations unless the user specifically asks for a detailed explanation. Do not use markdown, headings, bullet points, unnecessary formatting or emojis in normal conversation. These are MAX's saved memories about the user: ${memoryText} Use memories only when relevant. Do not mention the memory system unless the user asks. User: ${message}`; // ========================= // CLOUDFLARE AI // ========================= const response = await fetch( `https://api.cloudflare.com/client/v4/accounts/${ACCOUNT_ID}/ai/run/@cf/meta/llama-3.1-8b-instruct`, { method: "POST", headers: { "Authorization": `Bearer ${API_TOKEN}`, "Content-Type": "application/json" }, body: JSON.stringify({ prompt: prompt }) } ); const data = await response.json(); if (!response.ok || !data.success) { console.error( "Cloudflare AI Error:", data ); return res.status(500).json({ error: "MAX could not get a response from Cloudflare AI." }); } // ========================= // SEND REPLY // ========================= res.json({ reply: data.result.response }); } catch (error) { console.error( "Server Error:", error ); res.status(500).json({ error: "MAX server error." }); } }); // ========================= // START SERVER // ========================= const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log( `MAX backend running on port ${PORT}` ); });
